cmake_minimum_required(VERSION 3.21)
project(NextNotes VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 QuickDialogs2 Svg WebEngineCore WebEngineQuick WebChannel)
find_package(SQLite3 REQUIRED)
qt_standard_project_setup(REQUIRES 6.8)
set(KOS_UI_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/nextkdeAppDemo" CACHE PATH "Explicit development override for the pinned Kos.Ui source")
if(NOT EXISTS "${KOS_UI_SOURCE_DIR}/shared/qml/CMakeLists.txt")
    message(FATAL_ERROR "Initialize submodules or explicitly set KOS_UI_SOURCE_DIR")
endif()
add_subdirectory("${KOS_UI_SOURCE_DIR}/shared/qml" kos-ui)
qt_add_library(nextnotes-storage STATIC
    src/storage/MarkdownFile.cpp src/storage/MarkdownFile.h
    src/storage/NotesRepository.cpp src/storage/NotesRepository.h
    src/storage/ManagedStore.cpp src/storage/ManagedStore.h)
target_include_directories(nextnotes-storage PUBLIC src/storage)
target_link_libraries(nextnotes-storage PUBLIC Qt6::Core SQLite::SQLite3)
qt_add_executable(nextnotes src/main.cpp src/EditorSession.cpp src/EditorSessionWeb.cpp src/EditorSession.h src/NotesController.cpp src/NotesLibrary.cpp src/EditorExport.cpp src/PdfExport.cpp src/NotesOrganizer.cpp src/NotesKnowledge.cpp src/NotesController.h)
target_sources(nextnotes PRIVATE src/EditorRenderLifecycle.cpp src/EditorRenderLifecycle.h)
qt_add_qml_module(nextnotes URI NextNotes VERSION 1.0 RESOURCE_PREFIX /qt/qml
    QML_FILES qml/NotesSelectionBox.qml qml/FolderIcons.js qml/PopupPlacement.js qml/NotesPopover.qml qml/NotesContentPopover.qml qml/NotesSettings.qml qml/LibraryActions.qml qml/NotesWindowControls.qml qml/KnowledgePanel.qml qml/RecycleBin.qml qml/Main.qml qml/NotesOrganizer.qml qml/SourceEditor.qml qml/EditorTools.qml qml/CommandRegistry.qml qml/NotesTokens.qml qml/NoteReadingView.qml)
qt_add_resources(nextnotes window-controls PREFIX /nextnotes/platform BASE qml FILES qml/NativeWindowControls.qml)
file(GLOB_RECURSE EDITOR_ASSETS CONFIGURE_DEPENDS web/editor/dist/* web/editor/vendor/*)
qt_add_resources(nextnotes editor-assets PREFIX /editor BASE web/editor FILES web/editor/index.html web/editor/editor.css ${EDITOR_ASSETS})
target_link_libraries(nextnotes PRIVATE nextnotes-storage Kos::Ui Kos::UiPlugin Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::QuickDialogs2 Qt6::Svg Qt6::WebEngineCore Qt6::WebEngineQuick Qt6::WebChannel)
option(NEXTNOTES_BUILD_TESTS "Build editor contract tests (VM/CI only)" OFF)
if(NEXTNOTES_BUILD_TESTS)
    find_package(Qt6 6.8 REQUIRED COMPONENTS Test QuickTest)
    enable_testing()
    qt_add_executable(notes-chrome-test tests/chrome/main.cpp)
    qt_add_qml_module(notes-chrome-test URI NextNotes.ChromeTests VERSION 1.0 RESOURCE_PREFIX /qt/qml
        QML_FILES qml/NotesWindowControls.qml qml/NotesOrganizer.qml qml/FolderIcons.js qml/NotesTokens.qml
            qml/NotesPopover.qml qml/NotesContentPopover.qml qml/PopupPlacement.js)
    qt_add_resources(notes-chrome-test test-window-controls PREFIX /nextnotes/platform BASE qml FILES qml/NativeWindowControls.qml)
    target_compile_definitions(notes-chrome-test PRIVATE QUICK_TEST_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/chrome")
    target_link_libraries(notes-chrome-test PRIVATE Kos::Ui Kos::UiPlugin Qt6::QuickTest Qt6::QuickControls2)
    add_test(NAME notes-chrome COMMAND notes-chrome-test)
    set_tests_properties(notes-chrome PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
    qt_add_executable(editor-render-test tests/render/main.cpp src/EditorRenderLifecycle.cpp src/EditorRenderLifecycle.h)
    target_include_directories(editor-render-test PRIVATE src)
    target_compile_definitions(editor-render-test PRIVATE QUICK_TEST_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/tests/render")
    target_link_libraries(editor-render-test PRIVATE Qt6::QuickTest Qt6::WebEngineQuick)
    # Requires a VM graphical session; deliberately separate from offscreen CTest.
    qt_add_executable(notes-scale-validation tests/ScaleValidation.cpp)
    target_link_libraries(notes-scale-validation PRIVATE nextnotes-storage Qt6::Test)
    qt_add_executable(notes-repository-test tests/NotesRepositoryTest.cpp)
    target_link_libraries(notes-repository-test PRIVATE nextnotes-storage Qt6::Test)
    add_test(NAME notes-repository COMMAND notes-repository-test)
    qt_add_executable(notes-controller-test tests/NotesControllerTest.cpp src/NotesController.cpp src/NotesLibrary.cpp src/EditorExport.cpp src/PdfExport.cpp src/NotesOrganizer.cpp src/NotesKnowledge.cpp src/NotesController.h src/EditorSession.cpp src/EditorSessionWeb.cpp src/EditorSession.h)
    target_include_directories(notes-controller-test PRIVATE src)
    target_link_libraries(notes-controller-test PRIVATE nextnotes-storage Qt6::Test Qt6::Quick Qt6::Gui Qt6::WebEngineCore)
    add_test(NAME notes-controller COMMAND notes-controller-test)
    qt_add_executable(editor-session-test tests/EditorSessionTest.cpp src/EditorSession.cpp src/EditorSessionWeb.cpp src/EditorSession.h)
    target_include_directories(editor-session-test PRIVATE src)
    target_link_libraries(editor-session-test PRIVATE Qt6::Test Qt6::Quick Qt6::Gui)
    add_test(NAME editor-session COMMAND editor-session-test)
    set_tests_properties(editor-session notes-controller PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
include(GNUInstallDirs)
install(TARGETS nextnotes RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES web/editor/THIRD_PARTY_LICENSES.txt web/editor/package-lock.json DESTINATION ${CMAKE_INSTALL_DATADIR}/nextnotes/editor)

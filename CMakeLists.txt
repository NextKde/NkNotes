cmake_minimum_required(VERSION 3.21)
project(NextNotes VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 QuickDialogs2 Svg)
qt_standard_project_setup(REQUIRES 6.8)
set(KOS_UI_SOURCE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/nextkdeAppDemo" CACHE PATH "Explicit development override for the pinned Kos.Ui source")
if(NOT EXISTS "${KOS_UI_SOURCE_DIR}/shared/qml/CMakeLists.txt")
    message(FATAL_ERROR "Initialize submodules or explicitly set KOS_UI_SOURCE_DIR")
endif()
add_subdirectory("${KOS_UI_SOURCE_DIR}/shared/qml" kos-ui)
qt_add_executable(nextnotes src/main.cpp src/EditorSession.cpp src/EditorSession.h)
qt_add_qml_module(nextnotes URI NextNotes VERSION 1.0 RESOURCE_PREFIX /qt/qml
    QML_FILES qml/Main.qml qml/SourceEditor.qml qml/CommandRegistry.qml qml/NotesTokens.qml)
target_link_libraries(nextnotes PRIVATE Kos::Ui Kos::UiPlugin Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::QuickDialogs2 Qt6::Svg)
option(NEXTNOTES_BUILD_TESTS "Build editor contract tests (VM/CI only)" OFF)
if(NEXTNOTES_BUILD_TESTS)
    find_package(Qt6 6.8 REQUIRED COMPONENTS Test)
    enable_testing()
    qt_add_executable(editor-session-test tests/EditorSessionTest.cpp src/EditorSession.cpp src/EditorSession.h)
    target_include_directories(editor-session-test PRIVATE src)
    target_link_libraries(editor-session-test PRIVATE Qt6::Test Qt6::Quick Qt6::Gui)
    add_test(NAME editor-session COMMAND editor-session-test)
    set_tests_properties(editor-session PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
include(GNUInstallDirs)
install(TARGETS nextnotes RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})

cmake_minimum_required(VERSION 3.21)
project(EditorComparison LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 QuickDialogs2 Svg WebEngineQuick)
qt_standard_project_setup(REQUIRES 6.8)
add_subdirectory(../../third_party/nextkdeAppDemo/shared/qml kos-ui)
if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/web/dist/editor.js")
 message(FATAL_ERROR "Build web bundle in VM first (npm install then npm run build)")
endif()
qt_add_executable(editor-comparison main.cpp Native.h)
qt_add_qml_module(editor-comparison URI EditorComparison VERSION 1.0 RESOURCE_PREFIX /qt/qml QML_FILES Main.qml)
qt_add_resources(editor-comparison assets PREFIX /editor FILES fixture.md web/index.html web/dist/editor.js)
target_link_libraries(editor-comparison PRIVATE Kos::Ui Kos::UiPlugin Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::QuickDialogs2 Qt6::Svg Qt6::WebEngineQuick)
